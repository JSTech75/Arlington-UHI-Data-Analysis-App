# Python Practice no.1 : Bouncing Ball

ypos = 0
y_speed = 3

def setup():
    global ypos
    size(400,400)
    background(255,255,0)

def draw():
    global ypos, y_speed
    background(255,255,0)
    fill(255,0,0)
    circle(200, ypos, 25)
    ypos += y_speed
    
    if ypos >= 400 or ypos <= 0:
           ypos = -ypos
